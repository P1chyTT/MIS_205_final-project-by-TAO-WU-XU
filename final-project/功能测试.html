当前问题：新功能（图片上传、行程费用、联系方式）无法使用
根本原因：数据库迁移脚本未在Supabase中运行

🚀 解决方案

步骤1：运行数据库迁移

在Supabase SQL Editor中运行以下SQL：

-- 方法1：运行完整schema（推荐） -- 复制 final-project/schema.sql 的全部内容到SQL Editor并执行 -- 方法2：只运行新功能迁移 -- 复制以下SQL并执行： -- 1. 添加联系方式字段 ALTER TABLE posts ADD COLUMN IF NOT EXISTS contact_type VARCHAR(10), ADD COLUMN IF NOT EXISTS contact_info VARCHAR(100); -- 2. 创建行程费用表 CREATE TABLE IF NOT EXISTS trip_expenses ( id SERIAL PRIMARY KEY, post_id INT NOT NULL REFERENCES posts(id) ON DELETE CASCADE, name VARCHAR(100) NOT NULL, amount DECIMAL(10,2) NOT NULL, note TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 3. 创建索引 CREATE INDEX IF NOT EXISTS idx_trip_expenses_post_id ON trip_expenses(post_id); -- 4. 启用RLS ALTER TABLE trip_expenses ENABLE ROW LEVEL SECURITY; -- 5. RLS策略：所有参与者都可以管理费用 CREATE POLICY "参与者可管理费用" ON trip_expenses FOR ALL USING ( EXISTS ( SELECT 1 FROM participations p WHERE p.post_id = trip_expenses.post_id AND p.user_id = auth.uid() ) OR EXISTS ( SELECT 1 FROM posts WHERE posts.id = trip_expenses.post_id AND posts.user_id = auth.uid() ) ); -- 6. 添加图片字段（如果还没有） ALTER TABLE posts ADD COLUMN IF NOT EXISTS image_url TEXT, ADD COLUMN IF NOT EXISTS image_urls TEXT[] DEFAULT '{}';

步骤2：验证迁移成功

运行验证SQL：

-- 检查联系方式字段 SELECT column_name, data_type FROM information_schema.columns WHERE table_name = 'posts' AND column_name IN ('contact_type', 'contact_info', 'image_url', 'image_urls'); -- 检查费用表 SELECT table_name, column_name, data_type FROM information_schema.columns WHERE table_name = 'trip_expenses';
